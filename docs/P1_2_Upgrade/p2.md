## 3. DECISION ENGINE (P2A)

### 3.1 Concept

```
Decision Engine không thay thế Health Score.
Nó là một OVERLAY điều chỉnh signal.

Health Score = "Coin đang khỏe thế nào?"
Decision Signal = "Hôm nay tôi có nên trading không?"

Ví dụ:
  CARV: Health = 86 (HEALTHY)
  CARV Event Risk: Unlock tomorrow (CRITICAL)
  Decision: CAUTION (downgrade signal)

  AI16Z: Health = 72 (NEUTRAL)
  AI16Z Event Risk: None
  AI16Z Correlation: Low (diversified)
  Decision: WATCH (upgrade nếu derivatives strong)
```

### 3.2 Adjustment Formula

```typescript
interface DecisionInput {
  coinId:         number;
  date:           string;
  healthScore:    number;       // 0-100
  eventRiskScore: number;       // 0-100 (100=max risk)
  correlationRisk: number;      // 0-100 (100=highly correlated narrative)
}

function calculateAdjustedScore(input: DecisionInput): {
  adjustedScore:    number;
  adjustmentReason: string;
} {
  let score  = input.healthScore;
  let reason = '';

  // Event Risk Penalty
  if (input.eventRiskScore >= 80) {
    const penalty = 25;
    score -= penalty;
    reason += `Critical event risk (-${penalty}pts). `;
  } else if (input.eventRiskScore >= 60) {
    const penalty = 15;
    score -= penalty;
    reason += `High event risk (-${penalty}pts). `;
  } else if (input.eventRiskScore >= 40) {
    const penalty = 8;
    score -= penalty;
    reason += `Moderate event risk (-${penalty}pts). `;
  }

  // Correlation Risk (narrative concentration)
  if (input.correlationRisk >= 80) {
    const penalty = 10;
    score -= penalty;
    reason += `High narrative correlation (-${penalty}pts). `;
  }

  // Floor at 5 (never completely zero unless truly broken)
  score = Math.max(5, Math.min(100, score));

  return {
    adjustedScore:    Math.round(score * 100) / 100,
    adjustmentReason: reason.trim() || 'No risk adjustments applied',
  };
}
```

### 4.2 Event Risk Score Calculation

```typescript
function getCoinEventRiskScore(
  coinId: number,
  date: string,
  events: EventRisk[]
): number {

  // Filter active events for this coin on this date
  const activeEvents = events.filter(e =>
    e.coinId === coinId &&
    e.isActive &&
    e.eventDate <= date &&
    (!e.expiresAt || e.expiresAt >= date)
  );

  if (activeEvents.length === 0) return 0;

  // Max risk from any single event (not sum, to avoid inflation)
  const maxRisk = Math.max(...activeEvents.map(e => e.riskScore ?? 0));

  // Small bonus for multiple events (max +15)
  const multiEventBonus = Math.min((activeEvents.length - 1) * 5, 15);

  return Math.min(maxRisk + multiEventBonus, 100);
}

## 5. CORRELATION ANALYSIS (P2C)

### 5.1 Correlation Calculation

```typescript
function calculatePearsonCorrelation(
  series1: number[],  // daily health scores coin A
  series2: number[]   // daily health scores coin B
): number {
  // Standard Pearson correlation
  const n = Math.min(series1.length, series2.length);
  if (n < 5) return 0;  // Not enough data

  const mean1 = series1.slice(-n).reduce((a, b) => a + b) / n;
  const mean2 = series2.slice(-n).reduce((a, b) => a + b) / n;

  let num = 0;
  let den1 = 0;
  let den2 = 0;

  for (let i = 0; i < n; i++) {
    const d1 = series1[i] - mean1;
    const d2 = series2[i] - mean2;
    num  += d1 * d2;
    den1 += d1 * d1;
    den2 += d2 * d2;
  }

  const denom = Math.sqrt(den1 * den2);
  return denom === 0 ? 0 : num / denom;
}
```

### 5.2 Narrative Correlation Risk

```typescript
function getNarrativeCorrelationRisk(
  correlations: number[]   // all pairwise correlations within narrative
): number {
  if (correlations.length === 0) return 0;
  const avgCorrelation = correlations.reduce((a, b) => a + b) / correlations.length;
  // avgCorr 0 → risk 0, avgCorr 1 → risk 100
  return Math.round(Math.max(0, avgCorrelation) * 100);
}
```

## 6. NARRATIVE MOMENTUM (P2D)

### 6.1 Concept

```
Narrative Health Timeline:
  Day 1:  72
  Day 2:  74
  Day 3:  78
  Day 4:  81   ← Momentum ACCELERATING (slope increasing)
  Day 5:  83
  Day 6:  84
  Day 7:  84.5 ← Momentum DECELERATING (slope decreasing)

Momentum Score = rate of change of health score
  Positive → Narrative improving
  Negative → Narrative deteriorating
  Near zero → Stable or topping
```

### 6.2 Momentum Calculation

```typescript
function calculateNarrativeMomentum(
  healthHistory: Array<{ date: string; healthScore: number }>
): NarrativeMomentum {

  if (healthHistory.length < 3) {
    return { score: 0, type: 'stable', health7dAgo: null, healthNow: null };
  }

  const recent = healthHistory.slice(-7);
  const now    = recent[recent.length - 1].healthScore;
  const ago7d  = recent[0].healthScore;

  // First derivative (rate of change)
  const change7d = now - ago7d;

  // Second derivative (rate of rate of change)
  // Compare first half vs second half slope
  const midpoint = Math.floor(recent.length / 2);
  const firstHalfSlope  = linearSlope(recent.slice(0, midpoint));
  const secondHalfSlope = linearSlope(recent.slice(midpoint));
  const acceleration    = secondHalfSlope - firstHalfSlope;

  let type: 'accelerating' | 'decelerating' | 'stable';
  if      (Math.abs(acceleration) < 0.5) type = 'stable';
  else if (acceleration > 0)              type = 'accelerating';
  else                                    type = 'decelerating';

  // Momentum score: -100 to +100
  const momentumScore = Math.round(Math.max(-100, Math.min(100, change7d * 10)));

  return {
    score:       momentumScore,
    type,
    health7dAgo: ago7d,
    healthNow:   now,
  };
}

export const TRIGGER_TYPES = {
  SCORE_BELOW: {
    description: 'Health score drops below threshold',
    evaluate: (current: number, threshold: number) => current < threshold,
  },
  SCORE_DROP: {
    description: 'Health score drops by X points vs yesterday',
    evaluate: (change: number, threshold: number) => change < -threshold,
  },
  SCORE_RISE: {
    description: 'Health score rises by X points vs yesterday',
    evaluate: (change: number, threshold: number) => change > threshold,
  },
  SIGNAL_CHANGE: {
    description: 'Recommendation signal changes',
    evaluate: (prev: string, curr: string) => prev !== curr,
  },
  NARRATIVE_WEAK: {
    description: 'Narrative health drops below threshold',
    evaluate: (current: number, threshold: number) => current < threshold,
  },
} as const;
```

### 7.2 Alert Evaluation

```typescript
async function evaluateAlerts(
  date: string,
  coinScores: CoinScoreMap,
  previousScores: CoinScoreMap
): Promise<AlertFired[]> {

  const activeRules = await getActiveAlertRules();
  const fired: AlertFired[] = [];

  for (const rule of activeRules) {
    const current  = coinScores[rule.scopeId ?? 'global'];
    const previous = previousScores[rule.scopeId ?? 'global'];
    if (!current) continue;

    let triggered = false;
    switch (rule.triggerType) {
      case 'SCORE_BELOW':
        triggered = current.healthScore < rule.triggerValue;
        break;
      case 'SCORE_DROP':
        if (previous) {
          triggered = (current.healthScore - previous.healthScore) < -rule.triggerValue;
        }
        break;
      case 'SIGNAL_CHANGE':
        if (previous) {
          triggered = current.signal !== previous.signal;
        }
        break;
    }

    if (triggered) {
      fired.push({
        ruleId: rule.id,
        triggerDetail: { current, previous, threshold: rule.triggerValue },
      });
    }
  }

  return fired;
}
```
